  252    function _stake(address account, uint256 amount) internal {
  253      require(amount > 0, "AMOUNT_INVALID");
  254      stakes[account].duration = duration;
  255      if (stakes[account].balance == 0) {
  256        stakes[account].balance = amount;
  257        stakes[account].timestamp = block.timestamp;
  258      } else {
  259        uint256 nowAvailable = available(account);
  260        stakes[account].balance = stakes[account].balance.add(amount);
  261        stakes[account].timestamp = block.timestamp.sub(
  262          nowAvailable.mul(stakes[account].duration).div(stakes[account].balance)
  263        );
  264      }
  265      token.safeTransferFrom(msg.sender, address(this), amount);
  266      emit Transfer(address(0), account, amount);
  267    }
