  464          uint256 margin,
  465          uint256 fee,

  470                         
  471          if (pnlAfterFee < 0) {
  472              uint256 _pnlAfterFee = uint256(-1 * pnlAfterFee);
  473              if (_pnlAfterFee < margin) {
  474                  IERC20(token).uniTransfer(position.owner, (margin.sub(_pnlAfterFee)).mul(tokenBase).div(BASE));
  475                  vault.balance += uint96(_pnlAfterFee);
  476              } else {
  477                  vault.balance += uint96(margin);

  479              }
  480  
  481          } else {
  482              uint256 _pnlAfterFee = uint256(pnlAfterFee);
  483                            
  484              require(uint256(vault.balance) >= _pnlAfterFee, "!vault-insufficient");
  485              vault.balance -= uint96(_pnlAfterFee);
  486  
  487              IERC20(token).uniTransfer(position.owner, (margin.add(_pnlAfterFee)).mul(tokenBase).div(BASE));
  488          }
